cmake_minimum_required(VERSION 3.8)
project(scan_stereo_depth)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(message_filters REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core calib3d imgproc)
find_package(Threads REQUIRED)
add_executable(stereo_depth_node src/stereo_depth_node.cpp)
ament_target_dependencies(stereo_depth_node cv_bridge message_filters rclcpp sensor_msgs std_msgs)
target_include_directories(stereo_depth_node PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(stereo_depth_node ${OpenCV_LIBS} Threads::Threads)
install(TARGETS stereo_depth_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
